```
#!/bin/bash
# find-my-keys.sh
# 1) locate potential private-key files
# 2) mail you a concise report

#–– CONFIGURATION ––
EMAIL="you@example.com"            # where to send the report
TMP="/tmp/key-report-$$.txt"       # temp file for findings
SEARCH_ROOTS=("/home" "/root" "/etc" "/opt")  # where to look
PATTERNS=( "*.pem" "*.key" "id_rsa" "id_dsa" "*.ppk" )

#–– FIND STEP ––
> "$TMP"
for d in "${SEARCH_ROOTS[@]}"; do
  find "$d" -type f \( \
    $(printf -- "-iname \"%s\" -o " "${PATTERNS[@]}" | sed 's/ -o $//') \
  \) -perm /600 -print 2>/dev/null >> "$TMP"
done

#–– REPORT STEP ––
if [[ -s "$TMP" ]]; then
  mail -s "Key-File Report from $(hostname)" "$EMAIL" < "$TMP"
else
  # If you prefer only mail when keys are found, comment out the next line:
  echo "No candidate key files found on $(hostname) at $(date)." \
    | mail -s "Key-File Report (empty)" "$EMAIL"
fi

#–– CLEANUP ––
rm -f "$TMP"
```

```python
#!/usr/bin/env python3
import os, fnmatch, smtplib
from email.message import EmailMessage

EMAIL = "you@example.com"
HOST  = os.uname().nodename

# Where to search
SEARCH_ROOTS = ["/home", "/root", "/etc", "/opt"]
PATTERNS     = ["*.pem","*.key","id_rsa","id_dsa","*.ppk"]

found = []
for root in SEARCH_ROOTS:
    for dirpath, dirs, files in os.walk(root, onerror=lambda e: None):
        for p in PATTERNS:
            for fname in fnmatch.filter(files, p):
                full = os.path.join(dirpath, fname)
                if os.stat(full).st_mode & 0o600:
                    found.append(full)

msg = EmailMessage()
msg["From"] = EMAIL
msg["To"]   = EMAIL
msg["Subject"] = f"Key-File Report from {HOST}"
if found:
    msg.set_content("Found the following key files:\n\n" + "\n".join(found))
else:
    msg.set_content("No candidate key files found on " + HOST)

with smtplib.SMTP("localhost") as s:
    s.send_message(msg)
```

   Below is a Python script sketch using the Kubernetes API. Adapt it to your orchestration or cloud-native autoscaler (ECS, AKS, GKE, etc.):

```python
from kubernetes import client, config
import time

# CONFIGURE THESE
NAMESPACE          = "logistics"
DEPLOYMENT_NAME    = "agent-optimizer"
MAX_REPLICAS       = 20
MIN_REPLICAS       = 2
CPU_SCALE_UP_THRESH = 0.70  # 70%
CPU_SCALE_DOWN_THRESH = 0.40  # 40%
QUEUE_SCALE_UP_THRESH = 10000
QUEUE_SCALE_DOWN_THRESH = 3000
CHECK_INTERVAL_SEC = 60

def get_current_metrics():
    # Hook in to your monitoring system / message-queue stats
    cpu_usage = fetch_average_cpu_usage(DEPLOYMENT_NAME, NAMESPACE)
    queue_length = fetch_order_queue_length()
    return cpu_usage, queue_length

def fetch_current_replicas(api):
    resp = api.read_namespaced_deployment(DEPLOYMENT_NAME, NAMESPACE)
    return resp.spec.replicas

def scale_deployment(api, new_replicas):
    body = {'spec': {'replicas': new_replicas}}
    api.patch_namespaced_deployment(DEPLOYMENT_NAME, NAMESPACE, body)
    print(f"Scaled to {new_replicas} replicas")

def main():
    config.load_incluster_config()  # or load_kube_config()
    api = client.AppsV1Api()
    while True:
        cpu, queue_len = get_current_metrics()
        current = fetch_current_replicas(api)

        # Decide scale up
        if (cpu > CPU_SCALE_UP_THRESH or queue_len > QUEUE_SCALE_UP_THRESH) and current < MAX_REPLICAS:
            target = min(current + 2, MAX_REPLICAS)
            scale_deployment(api, target)

        # Decide scale down
        elif (cpu < CPU_SCALE_DOWN_THRESH and queue_len < QUEUE_SCALE_DOWN_THRESH) and current > MIN_REPLICAS:
            target = max(current - 1, MIN_REPLICAS)
            scale_deployment(api, target)

        time.sleep(CHECK_INTERVAL_SEC)

if __name__ == "__main__":
    main()
```
